"""Preserve the failed admission attempt and repeat its unchanged workload."""
from pathlib import Path
import hashlib
import json
import os
import runpy
import shutil
import subprocess

R = Path(__file__).parent
P = R.with_name('profile-review-20261018')
m = runpy.run_path(str(P/'run-wrapper-models-v2.py'))
cache = Path(r'C:\workspace\wt\freetoken-mats\flash-perf\artifacts\strataq2perf9-cp312-cu130-sm120\kernel-cache')
python = Path(r'C:\workspace\envs\siltok-speed\flash-perf9\Scripts\python.exe')
for name, value in dict(RUN=R, PY=python, CACHE=cache).items():
    m[name] = value
    m['run'].__globals__[name] = value
stem = 'source-http-on-performance-v2'
archive = R/'scope-v1/resource-admission-attempt-v1'
archive.mkdir(exist_ok=False)
files = list(R.glob(stem+'.*'))
assert len(files) == 5, files
inventory = []
for item in files:
    inventory.append(dict(name=item.name, sha256=hashlib.sha256(item.read_bytes()).hexdigest()))
    shutil.move(str(item), str(archive/item.name))
(archive/'README.json').write_text(json.dumps(dict(
    reason='Warmup rejected before model load: 540672 bytes below unchanged 15.25 GiB admission threshold.',
    retry='User authorized clearing background GPU users; WidgetBoard and Radeon control panel stopped. Same model, workload, resource estimate and safety margins.',
    files=inventory), indent=2)+'\n', encoding='utf-8')
os.environ.update(PYTHONUTF8='1', PYTHONDONTWRITEBYTECODE='1', FREETOKEN_DISABLE_JIT='1',
    FREETOKEN_KERNEL_CACHE_DIR=str(cache), SILITOK_FREETOKEN_KERNEL_CACHE_DIR=str(cache),
    TRITON_CACHE_DIR=r'C:\workspace\cache\triton\flash-review2b')
os.environ.pop('PYTHONPATH', None)
m['run'](stem, [python, '-B', m['TOOLS']/'run-http-performance.py',
    *m['shared'](m['SOURCE'], 'on', 4992), '--prompts-json', m['TOOLS']/'frozen-prompts.json',
    '--tokens', 512, '--repetitions', 3, '--output', R/(stem+'.json')])
report = json.loads((R/(stem+'.json')).read_text(encoding='utf-8'))
assert report['status'] == 'measured' and len(report['runs']) == 6
assert all(row['output_tokens'] == 512 for row in report['runs'])
head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=m['SOURCE'], text=True).strip()
assert head == '02074c99d8c4b922cb063cdb4bb84dfef8172ae6'
(R/'models-complete-v1.json').write_text(json.dumps(dict(status='passed', speed_commit=head,
    qualified_model_trees=8, final_throughput_outputs=12,
    prior_72_output_abba_reference=str(P/'wrapper-regression-followup-v1.json'),
    preserved_failed_admission=str(archive),
    note='Seven first-attempt probes plus unchanged-config final ON retry passed; initial ON warmup admission failure preserved separately. Explicit external materials; FreeToken9/native/AOT/model geometry unchanged.'))+'\n')
print('ALL_SOURCE_MODELS_PASSED', flush=True)
