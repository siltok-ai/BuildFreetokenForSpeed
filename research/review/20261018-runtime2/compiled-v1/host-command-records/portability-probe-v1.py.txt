"""Check the actual relocated embedded interpreter and its dependency origins."""
import argparse
import importlib
import json
from pathlib import Path
import sys

p=argparse.ArgumentParser()
p.add_argument('--app',type=Path,required=True)
p.add_argument('--output',type=Path,required=True)
a=p.parse_args()
assert not a.output.exists()
app=a.app.resolve()
python=app/'python'
assert Path(sys.executable).resolve()==python/'python.exe'
origins={}
for name in ('freetoken','torch','triton','tvm_ffi','transformers','tokenizers','safetensors','numpy','psutil'):
    module=importlib.import_module(name)
    origin=Path(module.__file__).resolve()
    assert origin.is_relative_to(python),(name,str(origin))
    origins[name]=str(origin)
assert all(Path(path).resolve().is_relative_to(python) for path in sys.path if path),sys.path
from silitok_speed.backends.freetoken import FreeTokenBackend,_worker_command
from silitok_speed.backends.freetoken_qwen36 import FreeTokenQwen36Backend,FreeTokenQwen38Backend
from silitok_speed.backends.qwen36 import Qwen36Backend
from silitok_speed.qwen36.freetoken_wire import FrameWriter as LegacyWriter
from silitok_speed.runtimes.freetoken.protocol import FrameWriter
assert FreeTokenBackend is FreeTokenQwen36Backend is FreeTokenQwen38Backend is Qwen36Backend
assert LegacyWriter is FrameWriter
command=_worker_command()
assert Path(command[0]).resolve()==python/'python.exe'
assert str(python/'Lib/site-packages') in command
assert not any(fragment in part.lower() for part in command for fragment in ('\\envs\\','\\wt\\','\\tools\\'))
import torch
assert not torch.cuda.is_initialized()
result=dict(status='passed',scope='same-host relocated/sanitized embedded runtime; not clean OS',
    python=sys.executable,sys_path=sys.path,dependency_origins=origins,worker_command=command,
    legacy_aliases=True,cuda_parent_uninitialized=True)
a.output.write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result),flush=True)
