"""Process-only loader experiment; do not change installed package files."""
from pathlib import Path
import json
import os
import runpy
import time
import torch
import freetoken.moe.q2_residency as residency

original=residency.load_compact_banks
records=[]


def timed_load(*args, **kwargs):
    """只在独立实验进程的加载作用域调整线程，结束或失败均恢复。"""
    previous=torch.get_num_threads()
    requested=int(os.environ['Q2_LOAD_THREADS'])
    start=time.perf_counter()
    cpu=time.process_time()
    torch.set_num_threads(requested)
    try:
        result=original(*args,**kwargs)
        records.append(dict(seconds=time.perf_counter()-start,cpu_seconds=time.process_time()-cpu,
            threads_before=previous,threads_during=torch.get_num_threads(),host_bytes=result[1].host_bytes,
            gpu_bank_bytes=result[1].gpu_bank_bytes))
        return result
    finally:
        torch.set_num_threads(previous)
        Path(os.environ['Q2_LOAD_RECEIPT']).write_text(json.dumps(dict(records=records,threads_after=torch.get_num_threads()),indent=2)+'\n')


residency.load_compact_banks=timed_load
runpy.run_path(str(Path(__file__).with_name('native-nsys-probe-v1.py')),run_name='__main__')
