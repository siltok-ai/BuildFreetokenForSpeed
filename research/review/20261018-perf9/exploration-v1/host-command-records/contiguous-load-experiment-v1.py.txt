"""Load the one-file prototype in this process; preserve installed wheel bytes."""
from pathlib import Path
import hashlib
import json
import os
import runpy
import time
import torch
import freetoken.moe.q2_residency as residency

candidate=Path(r'C:\workspace\build\freetoken-mats\perf9-20261018-prototype\build-source\python\freetoken\moe\q2_residency.py')
variant=os.environ['Q2_COPY_VARIANT']
if variant=='contiguous':
    residency.__file__=str(candidate)
    exec(compile(candidate.read_bytes(),str(candidate),'exec'),residency.__dict__)
loader=residency.load_compact_banks


def timed_load(*args,**kwargs):
    start=time.perf_counter();cpu=time.process_time()
    result=loader(*args,**kwargs)
    Path(os.environ['Q2_LOAD_RECEIPT']).write_text(json.dumps(dict(seconds=time.perf_counter()-start,
        cpu_seconds=time.process_time()-cpu,threads=torch.get_num_threads(),variant=variant,
        prototype_source=str(candidate) if variant=='contiguous' else None,
        prototype_sha256=hashlib.sha256(candidate.read_bytes()).hexdigest() if variant=='contiguous' else None,
        host_bytes=result[1].host_bytes,gpu_bank_bytes=result[1].gpu_bank_bytes),indent=2)+'\n')
    return result


residency.load_compact_banks=timed_load
runpy.run_path(str(Path(__file__).with_name('native-nsys-probe-v1.py')),run_name='__main__')
