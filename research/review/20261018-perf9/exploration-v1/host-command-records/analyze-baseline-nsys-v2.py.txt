"""Summarize frozen traces; timing is diagnostic, not throughput qualification."""
from pathlib import Path
import collections
import json
import sqlite3

ROOT = Path(__file__).parent


def analyze(path):
    with sqlite3.connect(f"file:{path.as_posix()}?mode=ro", uri=True) as db:
        db.text_factory = lambda b: b.decode('utf-8', errors='replace')
        names = dict(db.execute('SELECT id,value FROM StringIds'))
        kernels = db.execute('SELECT start,end,shortName FROM CUPTI_ACTIVITY_KIND_KERNEL ORDER BY start').fetchall()
        groups = collections.defaultdict(lambda: [0, 0])
        union = cursor = 0
        for start, end, name in kernels:
            groups[names[name]][0] += 1
            groups[names[name]][1] += end-start
            union += max(0, end-max(start, cursor))
            cursor = max(cursor, end)
        api = [dict(name=n, calls=c, ms_per_window=t) for n,c,t in db.execute(
            'SELECT s.value,count(*),sum(a.end-a.start)/64e6 FROM CUPTI_ACTIVITY_KIND_RUNTIME a JOIN StringIds s ON a.nameId=s.id GROUP BY a.nameId ORDER BY sum(a.end-a.start) DESC LIMIT 15')]
        copies = [dict(kind=k, calls=c, mib=b, ms=t) for k,c,b,t in db.execute(
            'SELECT copyKind,count(*),sum(bytes)/1048576.0,sum(end-start)/1e6 FROM CUPTI_ACTIVITY_KIND_MEMCPY GROUP BY copyKind')]
    return dict(windows=64,kernel_count=len(kernels),gpu_span_ms_per_window=(kernels[-1][1]-kernels[0][0])/64e6,
        gpu_union_ms_per_window=union/64e6,kernel_sum_ms_per_window=sum(e-s for s,e,_ in kernels)/64e6,
        kernels=[dict(name=n,calls=c,ms_per_window=t/64e6) for n,(c,t) in sorted(groups.items(),key=lambda x:-x[1][1])[:32]],
        apis=api,copies=copies)


result = {mode: analyze(ROOT/f'baseline-{mode}-nsys-v1.sqlite') for mode in ('off','on')}
result['strata_prior'] = analyze(Path(r'C:\workspace\runs\strata\flash-perf-20261016\trace-mtp-v3.sqlite'))
result['scope'] = '64 warmed steps/windows; trajectories, resident budgets and token counts per window differ; not an engine throughput ratio'
output = ROOT/'baseline-nsys-analysis-v2.json'
output.write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
for mode,value in result.items():
    if not isinstance(value,dict): continue
    print(mode,json.dumps({k:v for k,v in value.items() if k not in ('kernels','apis','copies')}))
    for row in value['kernels'][:12]: print(row)
    print('apis', value['apis'][:6])
    print('copies',value['copies'])
