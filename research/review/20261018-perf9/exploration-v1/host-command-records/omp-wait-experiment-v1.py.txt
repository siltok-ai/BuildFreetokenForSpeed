"""Measure inherited OpenMP waiting policy in a fresh process without engine edits."""
from pathlib import Path
import ctypes
import json
import os
import runpy
import time
import torch
import freetoken.moe.q2_residency as residency
loader=residency.load_compact_banks


def timed_load(*args,**kwargs):
    runtime=ctypes.CDLL(str(Path(torch.__file__).parent/'lib/libiomp5md.dll'))
    runtime.kmp_get_blocktime.argtypes=[]
    runtime.kmp_get_blocktime.restype=ctypes.c_int
    blocktime=runtime.kmp_get_blocktime()
    start=time.perf_counter();cpu=time.process_time()
    result=loader(*args,**kwargs)
    Path(os.environ['Q2_LOAD_RECEIPT']).write_text(json.dumps(dict(seconds=time.perf_counter()-start,
        cpu_seconds=time.process_time()-cpu,threads=torch.get_num_threads(),kmp_blocktime_ms=blocktime,
        omp_wait_policy=os.environ.get('OMP_WAIT_POLICY'),host_bytes=result[1].host_bytes,
        gpu_bank_bytes=result[1].gpu_bank_bytes),indent=2)+'\n')
    return result


residency.load_compact_banks=timed_load
runpy.run_path(str(Path(__file__).with_name('native-nsys-probe-v1.py')),run_name='__main__')
