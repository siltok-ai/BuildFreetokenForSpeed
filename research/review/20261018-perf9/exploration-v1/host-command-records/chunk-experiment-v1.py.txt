"""Process-only read-chunk experiment; keep loader and arithmetic unchanged."""
from pathlib import Path
import json
import os
import runpy
import time
import torch
import freetoken.moe.q2_residency as residency
import freetoken.moe.expert_pieces as pieces

loader=residency.load_compact_banks
iterator=pieces.iter_expert_pieces


def iter_chunks(*args,**kwargs):
    """只改变专家读取的临时块上限，仍读取完整原始Q2字节。"""
    return iterator(*args,**(kwargs|dict(chunk=int(os.environ['Q2_CHUNK_MIB'])<<20)))


def timed_load(*args,**kwargs):
    start=time.perf_counter();cpu=time.process_time()
    result=loader(*args,**kwargs)
    Path(os.environ['Q2_LOAD_RECEIPT']).write_text(json.dumps(dict(seconds=time.perf_counter()-start,
        cpu_seconds=time.process_time()-cpu,threads=torch.get_num_threads(),chunk_mib=int(os.environ['Q2_CHUNK_MIB']),
        host_bytes=result[1].host_bytes,gpu_bank_bytes=result[1].gpu_bank_bytes),indent=2)+'\n')
    return result


pieces.iter_expert_pieces=iter_chunks
residency.load_compact_banks=timed_load
runpy.run_path(str(Path(__file__).with_name('native-nsys-probe-v1.py')),run_name='__main__')
