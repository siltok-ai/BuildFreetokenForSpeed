"""Qualify newly derived FT9 source and ordinary wheel before binding Speed."""
from pathlib import Path
import json
import os
import subprocess
import time
R=Path(__file__).parent
P=R.with_name('profile-review-20261018')
E=R.with_name('explore-20261017-evening')
B=Path(r'C:\workspace\wt\freetoken-mats\flash-perf')
SOURCE=Path(r'C:\workspace\build\freetoken-mats\perf9-materials-01\candidate-source\build-source')
PY=Path(r'C:\workspace\envs\siltok-speed\flash-perf9\Scripts\python.exe')
CACHE=B/'artifacts/strataq2perf9-cp312-cu130-sm120/kernel-cache'
deadline=time.monotonic()+3600
while not (P/'copy-materials-ready-v1.json').exists():
    if time.monotonic()>deadline:raise TimeoutError('FT9 materials incomplete')
    time.sleep(2)
text=(E/'run-freetoken-tests-b.py').read_text(encoding='utf-8')
text=text.replace(r'C:\workspace\build\freetoken-mats\review2b-materials-01\candidate-source\build-source',str(SOURCE))
text=text.replace('strataq2review2b','strataq2perf9')
(R/'run-freetoken-tests-v1.py').write_text(text,encoding='utf-8')
text=(P/'native-nsys-probe-v1.py').read_text(encoding='utf-8')
text=text.replace(r'C:\workspace\envs\siltok-speed\flash-review2b\Lib\site-packages\freetoken',str(PY.parents[1]/'Lib/site-packages/freetoken'))
text=text.replace('0.1.3+speed.strataq2review2b','0.1.3+speed.strataq2perf9').replace('ordinary_installed_review2b_wheel_not_Speed_acceptance','ordinary_installed_perf9_wheel_not_Speed_acceptance')
(R/'native-probe-v1.py').write_text(text,encoding='utf-8')
env=os.environ.copy()
env.update(PYTHONUTF8='1',PYTHONDONTWRITEBYTECODE='1',FREETOKEN_DISABLE_JIT='1',FREETOKEN_KERNEL_CACHE_DIR=str(CACHE),
    TRITON_CACHE_DIR=r'C:\workspace\cache\triton\flash-perf9')
env.pop('PYTHONPATH',None)
completed=[]
for mode,args in [('source',['--source',str(SOURCE)]),('wheel',[])]:
    stem=f'freetoken-{mode}-tests-v1'
    print('START',stem,flush=True)
    with (R/f'{stem}.log').open('w',encoding='utf-8') as log:
        subprocess.run([str(PY),'-B',str(R/'run-freetoken-tests-v1.py'),*args],env=env,cwd=R,stdout=log,stderr=subprocess.STDOUT,check=True)
    completed.append(stem)
    print('PASS',stem,flush=True)
for mode,resident in [('off',5760),('on',4992)]:
    stem=f'native-quality-{mode}-v1'
    args=[str(PY),'-B',str(R/'native-probe-v1.py'),'--output',str(R/f'{stem}.json'),
        '--resident',str(resident),'--dynamic','1024','--prefill','1024','--state-cache-slots','1',
        '--mapped-embedding','--adaptive-owners','--adapt-every','32','--adapt-decay','.7','--suite','quality','--rebuild']
    if mode=='on':args+=['--mtp','--variant','compact-mtp','--draft-policy','paired_cjk']
    print('START',stem,flush=True)
    subprocess.run([str(PY),'-B',r'C:\workspace\runs\siltok-speed\review-20261017\consumer-tools\supervise-v2.py',
        '--prefix',str(R/stem),'--cwd',str(R),'--timeout','1500','--',*args],env=env,check=True,stdout=subprocess.DEVNULL)
    data=json.loads((R/f'{stem}.json').read_text(encoding='utf-8'))
    resource=json.loads((R/f'{stem}.resource.json').read_text())
    assert data['status']=='passed' and not resource['stop_reason'] and not resource['remaining_before_cleanup']
    completed.append(stem)
    print('PASS',stem,flush=True)
(R/'native-qualification-complete-v1.json').write_text(json.dumps(dict(status='passed',completed=completed,
    version='0.1.3+speed.strataq2perf9',scope='ordinary installed FT9; Speed and protected package qualification remain separate'),indent=2)+'\n')
