$PayloadRoot = 'C:\workspace\build\siltok-speed\perf9 moved-p01'
$ResolvedPayload = [IO.Path]::GetFullPath($PayloadRoot)
if (-not $ResolvedPayload.StartsWith('C:\workspace\build\siltok-speed\', [StringComparison]::OrdinalIgnoreCase)) { throw 'Output escaped build root' }
if (Test-Path -LiteralPath $ResolvedPayload) { throw 'Fresh output required' }

$Materials = Join-Path $MaterialsRepo 'artifacts\strataq2perf9-cp312-cu130-sm120'

$Release = Get-Content (Join-Path $SourceRoot 'silitok_speed\qwen36\release.json') -Raw | ConvertFrom-Json
if ($Selected.custom_version -ne $Release.version) { throw 'Material version differs' }
$Wheel = Join-Path $Materials $Release.wheel.path

$PreflightCache = 'C:\workspace\runs\siltok-speed\perf9-20261018\packaging-preflight-cache-v1'

foreach ($Pattern in @('cpuinfer_ext*.pyd','KTransformersOps*.pyd','vLLMMarlin*.pyd')) {
    $Files=Get-ChildItem -LiteralPath C:\workspace\mats\siltok-speed\kt -Filter $Pattern -File
    if (-not $Files) { throw "Missing extension $Pattern" }
    $Files | Get-FileHash -Algorithm SHA256 | Format-List

& $ReleasePython -I -B -m silitok_speed.qwen36.freetoken_release generate-manifest --kernel-cache $PreflightCache

& $ReleasePython -I -B -m silitok_speed.qwen36.freetoken_release self-check --kernel-cache $PreflightCache --pip-check --constraint-policy (Join-Path $SourceRoot 'packaging\windows\runtime-requirements.toml')
