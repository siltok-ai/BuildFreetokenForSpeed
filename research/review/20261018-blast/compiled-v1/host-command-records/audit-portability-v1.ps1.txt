param([Parameter(Mandatory)][string]$SourceRoot,[Parameter(Mandatory)][string]$AppRoot,[Parameter(Mandatory)][string]$Output)
$ErrorActionPreference='Stop'
# 只检查可复用打包代码和实际部署配置；构建来源记录允许保留本机输入路径。
$Tracked=git -C $SourceRoot ls-files packaging/windows
$CodeFiles=@($Tracked | Where-Object { $_ -match '\.(ps1|py|toml)$' } | ForEach-Object { Join-Path $SourceRoot $_ })
$DeployFiles=@('launch.cmd','SiltokSpeed.xml','managed-base.yaml','select-cpu-variant.ps1','python\python312._pth') | ForEach-Object { Join-Path $AppRoot $_ }
$Patterns=@('C:\workspace','C:\\workspace','C:/workspace','C:\Users\UserPC','C:\\Users\\UserPC')
$Findings=@();$SyntaxFindings=@()
foreach($File in ($CodeFiles+$DeployFiles)) {
    if(-not(Test-Path -LiteralPath $File)){throw "Missing scanned file: $File"}
    $Body=[IO.File]::ReadAllText($File)
    foreach($Pattern in $Patterns){if($Body.Contains($Pattern,[StringComparison]::OrdinalIgnoreCase)){$Findings+=[pscustomobject]@{path=$File;pattern=$Pattern}}}
    if($File.EndsWith('.ps1')) {
        $ParseTokens=$null;$ParseErrors=$null
        [Management.Automation.Language.Parser]::ParseFile($File,[ref]$ParseTokens,[ref]$ParseErrors) | Out-Null
        foreach($Problem in $ParseErrors){$SyntaxFindings+=[pscustomobject]@{path=$File;message=$Problem.Message}}
    }
}
$Record=[ordered]@{source_commit=(git -C $SourceRoot rev-parse HEAD);code_files=$CodeFiles.Count;deployment_files=$DeployFiles.Count;host_path_findings=$Findings;ps_syntax_findings=$SyntaxFindings}
if(Test-Path -LiteralPath $Output){throw 'Fresh report required'}
$Record | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath $Output -Encoding utf8
if($Findings.Count -or $SyntaxFindings.Count){throw 'Portability scan failed'}
