from pathlib import Path
import json
import subprocess
R=Path(__file__).parent
PY=r'C:\workspace\envs\siltok-speed\flash-perf9\Scripts\python.exe'
for name in ('run-wrapper-cpu-v2.py','prepare-runtime-release-v1.py','run-models-v2.py',
             'run-protected-build-v1.py','run-protected-acceptance-v1.py'):
    print('START',name,flush=True)
    with (R/(Path(name).stem+'.driver.log')).open('w',encoding='utf-8') as log:
        result=subprocess.run([PY,'-B',str(R/name)],stdout=log,stderr=subprocess.STDOUT)
    if result.returncode:
        (R/'validation-pipeline-failed-v1.json').write_text(json.dumps(dict(step=name,returncode=result.returncode))+'\n')
        raise SystemExit(result.returncode)
    print('PASS',name,flush=True)
(R/'validation-pipeline-complete-v1.json').write_text(json.dumps(dict(status='passed'))+'\n')
