"""遵循仓库逐文件行尾规则，撤回本验收脚本自己的过宽LF转换。"""
from pathlib import Path
import ast
import json
import subprocess

R = Path(__file__).parent
OLD = R.with_name('perf9-20261018')
S = Path(r'C:\workspace\wt\siltok-speed\rebase-dev-20261018')
C = Path(r'C:\workspace\wt\siltok-speed\blast relocated')
commit = '9d871c87031d9d6de4946cb76af1abf66ab18df5'
names = subprocess.check_output(['git', 'ls-files'], cwd=S, text=True).splitlines()
# v1只改写了自己核对过AST等价的已提交Python文件，其他文件未改写。
# 在恢复前再次确认工作树Python字节就是HEAD blob，避免覆盖任何新逻辑。
python_names = [n for n in names if n.endswith('.py')]
for name in python_names:
    canonical = subprocess.check_output(['git', 'show', 'HEAD:'+name], cwd=S)
    assert (S/name).read_bytes() == canonical, name
    assert (C/name).read_bytes() == canonical, name
for root in (S, C):
    assert subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip() == commit
    subprocess.run(['git', '-c', 'core.safecrlf=false', 'restore', '--source=HEAD', '--worktree', '--', '.'], cwd=root, check=True)
    assert not subprocess.check_output(['git', '-c', 'core.safecrlf=false', 'status', '--porcelain'], cwd=root, text=True).strip()
assert all((S/n).read_bytes() == (C/n).read_bytes() for n in python_names)
for name, data in [
    ('canonical-checkout-v3.json', dict(status='passed', canonical_checkout_attributes=True,
        corrected_attempt='prepare-validation-v1.py globally forced LF and was rejected; both owned trees restored with Git attributes before any model/build qualification')),
    ('build-source-byte-audit-v2.json', dict(status='passed', source=str(S), checkout=str(C), speed_commit=commit,
        all_tracked_python_equal=True, git_clean=True)),
]:
    (R/name).write_text(json.dumps(data, indent=2)+'\n')
script=(R/'prepare-validation-v1.py').read_text(encoding='utf-8')
exec('def transform'+script.split('def transform',1)[1])
