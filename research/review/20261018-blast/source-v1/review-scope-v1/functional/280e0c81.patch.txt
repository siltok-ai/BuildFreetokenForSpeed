+def _checkpoint(tmp_path, monkeypatch):

+    config = json.loads((Path(__file__).parents[1]/'fixtures/qwen38-flash-next-nvfp4-config.json').read_text(encoding='utf-8'))

+    receipts, identities = [], {}

+        path = tmp_path / Path(name).name

+            filename=name, path=path.name, size=len(data), sha256=digest, verified_mtime_ns=path.stat().st_mtime_ns))

+def request(client, payload):
+    """逐行消费真实 TCP SSE；以 usage 的实际长度计数，尊重 EOS 和终止原因。"""
+    started = time.perf_counter()
+    arrivals, parts, reasoning, chunks = [], [], [], []
+    done = 0
+    with client.stream('POST', '/v1/chat/completions', json=payload | {'stream': True}) as response:
+        if response.status_code != 200:
+            response.read()
+            raise RuntimeError(f'HTTP request failed: {response.status_code} {response.text[:500]}')
+        for line in response.iter_lines():
+            if line == 'data: [DONE]':
+                done += 1
+            elif line.startswith('data: '):
+                chunk = json.loads(line[6:])
+                chunks.append(chunk)
+                for choice in chunk.get('choices', []):
+                    delta = choice.get('delta', {})
+                    content = delta.get('content') or ''
+                    thought = delta.get('reasoning_content') or ''
+                    if content or thought:
+                        arrivals.append(time.perf_counter())
+                        parts.append(content)
+                        reasoning.append(thought)
+    usage = next((c['usage'] for c in reversed(chunks) if c.get('usage')), None)
+    finish = next((v['finish_reason'] for c in reversed(chunks) for v in c.get('choices', [])
+                   if v.get('finish_reason')), None)
+    if done != 1 or usage is None or not finish or len(arrivals) < 2:
+        diagnostic = dict(done=done, usage=usage, finish=finish, arrivals=len(arrivals), chunks=chunks)
+        raise RuntimeError('Incomplete real-socket SSE response: '+json.dumps(diagnostic,ensure_ascii=False))
+    count = usage['completion_tokens']
+    return dict(usage=usage, finish_reason=finish, output_tokens=count, done_events=done,
+        text=''.join(parts), reasoning_content=''.join(reasoning), content_chunk_count=len(arrivals),
+        arrival_offsets=[t-started for t in arrivals], ttft_ms=1000*(arrivals[0]-started),
