+@pytest.mark.parametrize("field", ["presence_penalty", "frequency_penalty", "min_p"])
+def test_supported_deferred_numeric_overflow_is_an_input_error(qwen36_client, field):
+    """允许某字段的catalog仍须拒绝超大整数，不能在float转换时崩溃。"""
+    from silitok_speed.sdk import Client, SilitokError
+
+    http, engine = qwen36_client
+    catalog = _qwen36_catalog()
+    catalog[0]["supported_request_fields"].append(field)
+    catalog[0]["unsupported_request_fields"].remove(field)
+    http.app.state.model_catalog = catalog
+    payload = {"model": "m1", "messages": [{"role": "user", "content": "hello"}],
+               field: 10**400}
+    response = http.post("/v1/chat/completions", json=payload)
+    assert response.status_code == 400
+    assert response.json()["error"]["code"] == "invalid_request"
+    assert response.json()["error"]["param"] == field
+    with pytest.raises(SilitokError) as failure:
+        Client(engine=engine, model_catalog=catalog).chat.completions.create(**payload)
+    assert failure.value.code == "invalid_request"
+    assert failure.value.param == field
+    assert engine.task_queue.all_tasks() == []
