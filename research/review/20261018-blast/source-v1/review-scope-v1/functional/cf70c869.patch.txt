+"""执行文档中的 manifest，防止可复制示例与真实发现合同再次脱节。"""
+from __future__ import annotations
+
+import json
+from pathlib import Path
+
+from silitok_speed import Engine, EngineConfig
+from silitok_speed.discovery import discover_and_register
+from silitok_speed.manifest_schema import load_bytes
+from tests.support.mocks import make_mock_hw
+
+
+def test_qwen36_readme_manifest_registers_the_model(tmp_path, monkeypatch):
+    """用真实解析器、磁盘检查与 Engine 注册文档示例，不加载模型或触碰 GPU。"""
+    import silitok_speed
+    import silitok_speed.license as license_module
+    from silitok_speed.backends.freetoken_qwen36 import FreeTokenQwen36Backend
+
+    readme = (
+        Path(__file__).resolve().parents[2] / "silitok_speed" / "qwen36" / "README.md"
+    ).read_text(encoding="utf-8")
+    manifest_block = readme.split("```json", 1)[1].split("```", 1)[0]
+    manifest = load_bytes(manifest_block.encode("utf-8"))
+    entry = manifest.models[0]
+    model_dir = tmp_path / entry.artifacts["weights"].path
+    model_dir.mkdir(parents=True)
+    (model_dir / "config.json").write_text(
+        json.dumps({"quantization_config": {"quant_method": "modelopt"}}),
+        encoding="utf-8",
+    )
+    (model_dir / "tokenizer.json").write_text("{}", encoding="utf-8")
+    (model_dir / "model.safetensors").write_bytes(b"test-weight-placeholder")
+    (tmp_path / "manifest.json").write_text(manifest_block, encoding="utf-8")
+
+    # 原生授权边界只在本测试内替换；发现及注册调用链保持真实。
+    token = b"qwen-manifest-test".ljust(32, b"0")
+    monkeypatch.setattr(license_module, "_native", None)
+    monkeypatch.setattr(silitok_speed, "_license_verify", lambda path: (token, b"12345678"), raising=False)
+    monkeypatch.setattr(silitok_speed, "_license_check", lambda value: value == token, raising=False)
+    monkeypatch.setattr(silitok_speed, "_license_id", lambda: "qwen-manifest-test", raising=False)
+    grant = license_module.verify("test-fixture").grant
+    engine = Engine(
+        EngineConfig(models_dir=str(tmp_path), result_store_dir=str(tmp_path / "results")),
+        hw=make_mock_hw(),
+        grant_provider=lambda: grant,
+    )
+
+    count, catalog = discover_and_register(engine, str(tmp_path))
+
+    assert count == 1
+    assert catalog[0]["id"] == entry.id
+    assert catalog[0]["max_output_tokens"] == 4096
+    assert catalog[0]["supports_multimodal"] is False
+    registered = engine.model_mgr.resolve(entry.id)
+    assert registered.backend_cls is FreeTokenQwen36Backend
+    assert Path(registered.spec.weights_path) == model_dir
+    assert registered.spec.quantization == "modelopt"
+    assert registered.config.offload is False
+    assert registered.config.extra == {}
