+def test_protocol_isolates_native_and_child_stdout(tmp_path):
+    """在真实子进程中验证CRT描述符、Win32句柄和后代日志都不污染协议。"""
+    import io
+    import os
+    from pathlib import Path
+    import subprocess
+    import textwrap
+
+    from silitok_speed.qwen36.freetoken_wire import read_frame
+
+    package_root = Path(worker_module.__file__).resolve().parents[2]
+    child_code = "import os; os.write(1, b'child-stdout-marker\\n')"
+    script = '\n'.join([
+        'import os, sys, subprocess',
+        f'sys.path.insert(0, {str(package_root)!r})',
+        'from silitok_speed.qwen36.freetoken_worker import _redirect_stdout_to_protocol',
+        'from silitok_speed.qwen36.freetoken_wire import FrameWriter',
+        'protocol = _redirect_stdout_to_protocol()',
+        "print('python-stdout-marker', flush=True)",
+        "os.write(1, b'__TR_NATIVE_STDOUT_MARKER__\\n')",
+        f"subprocess.run([sys.executable, '-I', '-c', {child_code!r}], check=True)",
+    ])
+    if os.name == 'nt':
+        script += '\n' + textwrap.dedent('''
+            import ctypes
+            kernel = ctypes.WinDLL('kernel32', use_last_error=True)
+            kernel.GetStdHandle.argtypes = [ctypes.c_ulong]
+            kernel.GetStdHandle.restype = ctypes.c_void_p
+            kernel.WriteFile.argtypes = [ctypes.c_void_p, ctypes.c_char_p, ctypes.c_ulong,
+                                        ctypes.POINTER(ctypes.c_ulong), ctypes.c_void_p]
+            kernel.WriteFile.restype = ctypes.c_int
+            payload = b'win32-stdout-marker\\n'
+            written = ctypes.c_ulong()
+            if not kernel.WriteFile(kernel.GetStdHandle(-11), payload, len(payload), ctypes.byref(written), None):
+                raise ctypes.WinError(ctypes.get_last_error())
+        ''')
+    script += "\nFrameWriter(protocol).send({'ok': True, 'private_protocol': not os.get_inheritable(protocol.fileno())})\n"
+    result = subprocess.run([sys.executable, '-I', '-c', script], cwd=tmp_path,
+                            capture_output=True, check=True, timeout=30)
+    stream = io.BytesIO(result.stdout)
+    frame = read_frame(stream)
+    assert frame['ok'] and frame['private_protocol']
+    assert read_frame(stream) is None
+    for marker in (b'python-stdout-marker', b'__TR_NATIVE_STDOUT_MARKER__', b'child-stdout-marker'):
+        assert marker in result.stderr
+    if os.name == 'nt':
+        assert b'win32-stdout-marker' in result.stderr
+
