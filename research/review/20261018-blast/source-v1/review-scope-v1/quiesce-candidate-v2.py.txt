"""仅暂停本轮派发器；当前模型自然退出后停止父队列，保留原始证据。"""
from pathlib import Path
import json
import time
import psutil

R=Path(__file__).parent
selected=[]
for name in ('run-models-v2.py','run-validation-pipeline-v1.py'):
    matches=[]
    for p in psutil.process_iter(['pid','cmdline']):
        cmd=p.info['cmdline'] or []
        if any(str(R/name).lower()==str(part).lower() for part in cmd):matches.append(p)
    matching_ids={p.pid for p in matches}
    # Windows venv launcher和实际解释器拥有相同argv，只暂停真正运行派发代码的内层进程。
    process,=[p for p in matches if not any(c.pid in matching_ids for c in p.children())]
    process.suspend()
    selected.append(process)
active=selected[0].children(recursive=True)
(R/'paused-candidate-dispatchers-v2.json').write_text(json.dumps(dict(dispatchers=[p.pid for p in selected],
    model_descendants_allowed_to_finish=[p.pid for p in active]),indent=2)+'\n')
deadline=time.monotonic()+1200
while any(p.is_running() for p in active):
    if time.monotonic()>deadline:raise TimeoutError('Current model did not naturally finish')
    time.sleep(1)
path=R/'source-quality-on-v2.resource.json'
data=json.loads(path.read_text())
assert data['returncode']==0 and not data['stop_reason'] and not data['remaining_before_cleanup'] and not data['remaining_descendants']
for process in selected:
    assert not process.children(recursive=True)
    process.terminate()
psutil.wait_procs(selected,timeout=10)
(R/'candidate-quiescent-v1.json').write_text(json.dumps(dict(status='passed',candidate='9d871c87',
    current_model_natural_exit=True,dispatchers_stopped_after_child_completion=True,
    reason='Additional same-purpose public backend numeric-overflow fix; provisional qualification is superseded, no forced model termination.'))+'\n')
print('Current model naturally exited; only the two owned dispatchers stopped.',flush=True)
