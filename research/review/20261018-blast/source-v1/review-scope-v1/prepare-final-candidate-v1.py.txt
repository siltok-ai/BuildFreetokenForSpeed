"""按checkout实际行尾准备最终review候选；保留前一候选的独立原始记录。"""
from pathlib import Path
import ast
import json
import shutil
import subprocess

OLD=Path(__file__).parent
R=OLD.with_name('blast-final-20261018');R.mkdir(exist_ok=False)
S=Path(r'C:\workspace\wt\siltok-speed\rebase-dev-20261018')
C=Path(r'C:\workspace\wt\siltok-speed\blast final')
commit='4de927269439c59640578efcd46a10c001b65f72'
assert json.loads((OLD/'candidate-quiescent-v1.json').read_text())['status']=='passed'
assert subprocess.check_output(['git','rev-parse','HEAD'],cwd=S,text=True).strip()==commit
assert not subprocess.check_output(['git','status','--porcelain'],cwd=S,text=True).strip()
subprocess.run(['git','worktree','add','--detach',str(C),commit],cwd=S,check=True)
subprocess.run(['git','submodule','update','--init','--recursive','third_party/TSS.MSR'],cwd=C,check=True)
names=subprocess.check_output(['git','ls-files'],cwd=S,text=True).splitlines()
aligned=[]
for name in names:
    if not (name.endswith('.py') or name.startswith('silitok_speed/')):continue
    left=(S/name).read_bytes();right=(C/name).read_bytes()
    if left!=right:
        assert left.replace(b'\r\n',b'\n')==right.replace(b'\r\n',b'\n'),name
        if name.endswith('.py'):assert ast.dump(ast.parse(left))==ast.dump(ast.parse(right)),name
        shutil.copy2(C/name,S/name)
        aligned.append(name)
if aligned:subprocess.run(['git','-c','core.safecrlf=false','add','--',*aligned],cwd=S,check=True)
assert subprocess.run(['git','diff','--cached','--quiet'],cwd=S).returncode==0
for root in (S,C):assert not subprocess.check_output(['git','status','--porcelain'],cwd=root,text=True).strip()
assert all((S/n).read_bytes()==(C/n).read_bytes() for n in names if n.endswith('.py') or n.startswith('silitok_speed/'))
for name,data in [
    ('canonical-checkout-v3.json',dict(status='passed',canonical_checkout_attributes=True,aligned_eol_only=aligned,
        ast_unchanged=True,git_clean=True,prior_attempts=str(OLD))),
    ('build-source-byte-audit-v2.json',dict(status='passed',source=str(S),checkout=str(C),speed_commit=commit,
        all_tracked_python_equal=True,all_product_files_equal=True,git_clean=True)),
]: (R/name).write_text(json.dumps(data,indent=2)+'\n')

def copy(name,replacements=()):
    text=(OLD/name).read_text(encoding='utf-8')
    for old,new in replacements:
        assert old in text,(name,old)
        text=text.replace(old,new)
    (R/name).write_text(text,encoding='utf-8')

for name in ('run-eval-tests.py','run-wrapper-cpu-v2.py','backend8-ordinary-probe-v1.py',
             'run-http-compiled-alias-v1.py','portability-probe-v1.py','audit-portability-v1.ps1',
             'run-validation-pipeline-v1.py'):copy(name)
copy('prepare-runtime-release-v1.py',[
    ('flash-blast-release','flash-blast-final-release'),('blast-wheel-src','blast-final-wheel-src')])
copy('run-models-v2.py',[
    ('flash-blast-release','flash-blast-final-release'),
    ('9d871c87031d9d6de4946cb76af1abf66ab18df5',commit)])
copy('build-blast-p01.ps1',[
    ('-Task blast','-Task blast-final'),('blast relocated','blast final'),
    ('flash-blast-release','flash-blast-final-release'),('blast moved-p01','blast final-p01'),
    ('runs\\siltok-speed\\blast-review-20261018','runs\\siltok-speed\\blast-final-20261018')])
copy('run-protected-build-v1.py',[
    ('blast relocated','blast final'),('blast moved-p01','blast final-p01'),
    ('9d871c87031d9d6de4946cb76af1abf66ab18df5',commit)])
copy('run-protected-acceptance-v1.py',[
    ('blast moved-p01','blast final-p01'),('blast moved-test','blast final-test'),
    ('staged-blast-relocated','staged-blast-final')])
(R/'candidate-v1.json').write_text(json.dumps(dict(speed_commit=commit,speed_version='0.2.0+strataq2perf9.review2',
    source=str(S),checkout=str(C),ordinary_runtime=r'C:\workspace\envs\siltok-speed\flash-blast-final-release',
    freetoken_unchanged='0.1.3+speed.strataq2perf9'),indent=2)+'\n')
print('Final review candidate ready:',commit,'aligned',len(aligned),'line-ending-only files',flush=True)
