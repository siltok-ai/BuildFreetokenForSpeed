+def normalize_completion_token_budget(
+    values: Mapping[str, Any],
+) -> tuple[int | None, str | None]:
+    """把两个 OpenAI 输出预算别名归一为一个正整数。"""
+
+    max_tokens = values.get("max_tokens")
+    max_completion_tokens = values.get("max_completion_tokens")
+    if max_tokens is not None and max_completion_tokens is not None:
+        raise ChatContractError(
+            "Specify only one of max_tokens and max_completion_tokens.",
+            code="conflicting_parameters",
+            param="max_completion_tokens",
+        )
+    param = "max_tokens" if max_tokens is not None else "max_completion_tokens"
+    budget = max_tokens if max_tokens is not None else max_completion_tokens
+    if budget is None:
+        return None, None
+    if isinstance(budget, bool) or not isinstance(budget, int):
+        raise ChatContractError(
+            f"{param} must be an integer.",
+            code="invalid_parameter",
+            param=param,
+        )
+    if budget <= 0 or budget > MAX_CHAT_COMPLETION_TOKENS:
+        raise ChatContractError(
+            f"{param} must be between 1 and {MAX_CHAT_COMPLETION_TOKENS}.",
+            code="invalid_parameter",
+            param=param,
+        )
+    return budget, param

+def validate_model_fields(values: dict, capability: dict | None) -> None:
+    """按 discovery catalog 拒绝模型首版明确延期的请求能力。"""
+    messages = values.get("messages")
+    if (
+        not isinstance(messages, list)
+        or not messages
+        or any((not isinstance(message, dict) for message in messages))
+    ):
+        raise ChatContractError(

+            code="invalid_request",
+            param="messages",
+        )
+    for index, message in enumerate(messages):

+            raise ChatContractError(
+                f"messages[{index}].role must be a string.",
+                code="invalid_request",
+                param=f"messages.{index}.role",
+            )
+        content = message.get("content")
+        if content is not None and not isinstance(content, (str, list)):
+            raise ChatContractError(
+                f"messages[{index}].content must be text, an array, or null.",
+                code="invalid_request",
+                param=f"messages.{index}.content",
+            )
+        if isinstance(content, list) and any(
+            not isinstance(part, dict)

+            for part in content
+        ):
+            raise ChatContractError(
+                f"messages[{index}].content contains invalid content parts.",
+                code="invalid_request",
+                param=f"messages.{index}.content",
+            )
+    if capability is None:
+        return
+    has_supported_contract = "supported_request_fields" in capability
+    supported = set(capability.get("supported_request_fields") or [])
+    unsupported = set(capability.get("unsupported_request_fields") or [])

+    supplied = {
