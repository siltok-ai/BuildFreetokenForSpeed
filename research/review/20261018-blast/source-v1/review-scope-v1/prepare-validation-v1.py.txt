from pathlib import Path
import ast
import hashlib
import json
import subprocess

R = Path(__file__).parent
OLD = R.with_name('perf9-20261018')
S = Path(r'C:\workspace\wt\siltok-speed\rebase-dev-20261018')
C = Path(r'C:\workspace\wt\siltok-speed\blast relocated')
commit = '9d871c87031d9d6de4946cb76af1abf66ab18df5'
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=S, text=True).strip() == commit
assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=S, text=True).strip()
subprocess.run(['git', 'worktree', 'add', '--detach', str(C), commit], cwd=S, check=True)
subprocess.run(['git', 'submodule', 'update', '--init', '--recursive', 'third_party/TSS.MSR'], cwd=C, check=True)
names = subprocess.check_output(['git', 'ls-files'], cwd=S, text=True).splitlines()
normalized = []
for name in names:
    if not name.endswith('.py'):
        continue
    canonical = subprocess.check_output(['git', 'show', 'HEAD:'+name], cwd=S)
    for root in (S, C):
        before = (root/name).read_bytes()
        if before != canonical:
            assert before.replace(b'\r\n', b'\n') == canonical, name
            assert ast.dump(ast.parse(before)) == ast.dump(ast.parse(canonical)), name
            (root/name).write_bytes(canonical)
            normalized.append([str(root), name])
for root in (S, C):
    subprocess.run(['git', 'update-index', '--refresh'], cwd=root, check=True)
    assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=root, text=True).strip()
assert all((S/n).read_bytes() == (C/n).read_bytes() for n in names if n.endswith('.py'))
for name, data in [
    ('canonical-checkout-v3.json', dict(status='passed', canonical_from_head_blob=True, normalized=normalized)),
    ('build-source-byte-audit-v2.json', dict(status='passed', source=str(S), checkout=str(C), speed_commit=commit,
        all_tracked_python_equal=True, git_clean=True)),
]:
    (R/name).write_text(json.dumps(data, indent=2)+'\n')

def transform(source, target=None, replacements=()):
    text = (OLD/source).read_text(encoding='utf-8')
    for old, new in replacements:
        assert old in text, (source, old)
        text = text.replace(old, new)
    (R/(target or source)).write_text(text, encoding='utf-8')

for name in ('run-eval-tests.py', 'run-wrapper-cpu-v2.py', 'backend8-ordinary-probe-v1.py',

    transform(name)
